    import pandas as pd

df = pd.read_excel("sales.xlsx")
df["Name"] = df["Name"].str.strip().str.title()
df["Phone"] = (df["Phone"].astype(str)
    .str.replace(r"\D", "", regex=True).str[-10:])
df["Date"] = pd.to_datetime(df["Date"], dayfirst=True,
    format="mixed").dt.strftime("%d-%b-%Y")
df["Amount"] = pd.to_numeric(df["Amount"].astype(str)
    .str.extract(r"(\d[\d,]*\.?\d*)")[0]
    .str.replace(",", "", regex=False), errors="coerce")
df = df.drop_duplicates(subset=["Name", "Phone"])

df.to_excel("sales_clean.xlsx", index=False)
print(df[df["Amount"].isna()])  # rows with a missing amount
